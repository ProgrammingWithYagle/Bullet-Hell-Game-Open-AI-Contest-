import {availableParallelism} from 'node:os';
import {spawnSync} from 'node:child_process';
import {readFileSync,readdirSync,statSync} from 'node:fs';
import {writeTextAtomic as writeFileSync} from './write-text-atomic.mjs';
import {createHash} from 'node:crypto';
const walk=dir=>readdirSync(dir).sort().flatMap(name=>{const path=dir+'/'+name;return statSync(path).isDirectory()?walk(path):[path];});
const tests=readdirSync('tests').filter(n=>n.endsWith('.test.ts')).sort().map(n=>'tests/'+n);
// A bounded floor lets real peer/worker clocks overlap on small CI runners without extending any timeout or omitting a case.
const testFileConcurrency=Math.max(4,Math.min(8,availableParallelism()));
const commands=[['Stage 1 authored content',['tools/verify-stage1.mjs']],['Playable main-story word count',['tools/story-word-count.mjs','--check']],['TypeScript',['node_modules/typescript/bin/tsc','--noEmit']],['Early chapter build feasibility',['--import','tsx','tools/measure-opening-chapter.mts']],['Headless rules and real-network integration',['--import','tsx','--test','--test-concurrency='+testFileConcurrency,...tests]],['Production client build',['node_modules/vite/bin/vite.js','build','--logLevel','warn']],['Emitted encounter atlas contracts',['--import','tsx','tools/verify-encounter-asset-build.mjs']]];
const checks=[];for(const [name,args]of commands){const start=Date.now();console.log('\n'+name);const result=spawnSync(process.execPath,args,{stdio:'inherit',env:process.env});checks.push({name,passed:result.status===0,elapsedMs:Date.now()-start});if(result.status!==0){writeFileSync('reviews/runtime-verification.json',JSON.stringify({createdAt:new Date().toISOString(),passed:false,failedCheck:name,checks},null,2)+'\n');console.error(result.error??'Verification failed');process.exit(result.status??1);}}
const files=[...walk('src'),...walk('tests'),...['package.json','pnpm-lock.yaml','pnpm-workspace.yaml','tsconfig.json','vite.config.ts','index.html']];const hash=createHash('sha256');for(const file of files.sort()){hash.update(file+'\0');hash.update(readFileSync(file));}
const runtimeHash=createHash('sha256');for(const file of walk('src').sort()){runtimeHash.update(file+'\0');runtimeHash.update(readFileSync(file));}
const output=walk('dist').map(path=>({path,bytes:statSync(path).size,sha256:createHash('sha256').update(readFileSync(path)).digest('hex')}));
writeFileSync('reviews/runtime-verification.json',JSON.stringify({createdAt:new Date().toISOString(),passed:true,testFileConcurrency,testFiles:tests.length,verificationDriverSha256:createHash('sha256').update(readFileSync('tools/verify.mjs')).digest('hex'),sourceSnapshotSha256:hash.digest('hex'),runtimeSourceSha256:runtimeHash.digest('hex'),contentRevision:JSON.parse(readFileSync('content/revision.json')).sha256,checks,outputFiles:output.length,outputBytes:output.reduce((n,f)=>n+f.bytes,0),artifacts:output,limits:['Automated checks only; assistant browser observations and rendering samples are separate files under reviews/ui and reviews/performance.','No physical-device or human playtest is claimed.','All sixty-six boss identities execute; complete directional/action art, final balance, verified deployed leaderboards/recovery and final Windows/release acceptance remain open.']},null,2)+'\n');
console.log('\nVerification passed; evidence: reviews/runtime-verification.json');
