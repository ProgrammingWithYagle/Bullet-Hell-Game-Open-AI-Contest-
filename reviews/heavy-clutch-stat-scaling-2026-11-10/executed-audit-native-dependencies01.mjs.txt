import fs from 'node:fs';import path from 'node:path';import{fileURLToPath}from'node:url';import{createHash}from'node:crypto';
const stage=path.dirname(fileURLToPath(import.meta.url)),prior=path.resolve(stage,'../heavy-clutch-handling-movement-stage'),sha=b=>createHash('sha256').update(b).digest('hex');
const files=fs.readdirSync(stage).filter(n=>n==='native-probe.mts'||n==='earned-menu-flow.mts'||n.startsWith('retained-')&&n.endsWith('.mts'));
const missing=[];
for(const name of files){const s=fs.readFileSync(stage+'/'+name,'utf8'),refs=[...s.matchAll(/(?:from\s*|import\s*\()\s*(['"])(\.\/[^'"]+\.(?:mts|mjs|json))\1/g)].map(m=>m[2]);for(const m of s.matchAll(/new URL\(\s*(['"])([^'"]+\.(?:mts|mjs|json))\1\s*,\s*import.meta.url/g))refs.push(m[2]);for(const ref of refs){const target=path.resolve(stage,ref);if(!fs.existsSync(target))missing.push({importer:name,reference:ref,expectedFile:path.basename(target),exactRetainedSourceAvailable:fs.existsSync(prior+'/'+path.basename(target)),retainedSourceSha256:fs.existsSync(prior+'/'+path.basename(target))?sha(fs.readFileSync(prior+'/'+path.basename(target))):null});}}
const result={checkedAt:new Date().toISOString(),readOnly:true,driverSha256:sha(fs.readFileSync(stage+'/native-probe.mts')),filesInspected:files.map(file=>({file,sha256:sha(fs.readFileSync(stage+'/'+file))})),missing,canonicalFilesWritten:false,driverBytesChanged:false,repairExecuted:false};
fs.writeFileSync(stage+'/native-dependency-audit01.json',JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({missing,readOnly:true,driverBytesUnchanged:true}));
